import matplotlib.pyplot as plt
import numpy as np

# Get top 10 highest paying jobs
avg_salary = df.groupby('JOB_TITLE')['SALARY_USD'].mean().sort_values(ascending=False)
top10_jobs = avg_salary.head(10).sort_values()

# Create distinct colors (using colormap)
colors = plt.cm.tab10(range(len(top10_jobs)))

# Plot
plt.figure(figsize=(10,6))
bars = plt.barh(top10_jobs.index, top10_jobs.values, color=colors)

# Add value labels
for bar in bars:
    width = bar.get_width()
    plt.text(width, bar.get_y() + bar.get_height()/2,
             f'{width:,.0f}', va='center')

# Set X-axis ticks (every 75,000)
max_salary = top10_jobs.max()
ticks = np.arange(0, max_salary + 75000, 75000)
plt.xticks(ticks)

# Titles and labels
plt.title("Top 10 Highest Paying Job Roles", fontsize=14, fontweight='bold')
plt.xlabel("Average Salary (USD)", fontsize=12)
plt.ylabel("Job Title", fontsize=12)

plt.grid(axis='x', linestyle='--', alpha=0.7)
plt.tight_layout()
plt.show()

import pandas as pd

df['SALARY_USD'] = pd.to_numeric(df['SALARY_USD'], errors='coerce')
df['AUTOMATION_RISK_SCORE'] = pd.to_numeric(df['AUTOMATION_RISK_SCORE'], errors='coerce')

In [56]:
X = df_reg[['AUTOMATION_RISK_SCORE']]   # independent variable
y = df_reg['SALARY_USD']                # dependent variable
In [57]:
from sklearn.linear_model import LinearRegression

model = LinearRegression()
model.fit(X, y)

predictions = model.predict(X)
In [58]:
import matplotlib.pyplot as plt

plt.figure(figsize=(7,5))

plt.scatter(X, y, alpha=0.5)

plt.title("Automation Risk vs Salary (Linear Regression)", fontweight='bold')
plt.xlabel("Automation Risk Score")
plt.ylabel("Salary (USD)")

plt.grid(True)
plt.show()
